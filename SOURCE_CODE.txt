/// Empat atribut yang diminta dalam soal.
class Cryptocurrency {
  final int rank;
  final String name;
  final String symbol;
  // Simpan teks asli agar harga sangat kecil tidak dibulatkan menjadi 0.00.
  final String priceUsd;

  const Cryptocurrency({
    required this.rank,
    required this.name,
    required this.symbol,
    required this.priceUsd,
  });

  factory Cryptocurrency.fromJson(Map<String, dynamic> json) {
    final rank = int.tryParse(json['rank'].toString());
    final name = json['name'];
    final symbol = json['symbol'];
    final price = json['price_usd']?.toString();
    final numericPrice = price == null ? null : double.tryParse(price);
    if (rank == null || rank < 1 || name is! String || name.isEmpty ||
        symbol is! String || symbol.isEmpty || numericPrice == null ||
        !numericPrice.isFinite || numericPrice < 0) {
      throw const FormatException('Data cryptocurrency tidak valid.');
    }
    return Cryptocurrency(
      rank: rank,
      name: name,
      symbol: symbol,
      priceUsd: price!,
    );
  }
}

import 'dart:async';
import 'package:flutter/material.dart';
import '../models/cryptocurrency.dart';
import '../services/crypto_service.dart';

class CryptoPage extends StatefulWidget {
  const CryptoPage({super.key});

  @override
  State<CryptoPage> createState() => _CryptoPageState();
}

class _CryptoPageState extends State<CryptoPage> {
  final _service = CryptoService();
  List<Cryptocurrency> _coins = [];
  bool _loading = false;
  String? _error;

  @override
  void initState() {
    super.initState();
    _loadCoins();
  }

  Future<void> _loadCoins() async {
    if (_loading) return;
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      final coins = await _service.fetchCoins();
      if (!mounted) return;
      setState(() => _coins = coins);
    } on TimeoutException {
      if (!mounted) return;
      setState(() => _error = 'Waktu koneksi habis. Silakan coba lagi.');
    } on FormatException {
      if (!mounted) return;
      setState(() => _error = 'Format data API tidak dapat dibaca.');
    } catch (_) {
      if (!mounted) return;
      setState(() => _error = 'Gagal memuat data. Periksa internet, lalu tekan Refresh.');
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  @override
  void dispose() {
    _service.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Center(
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 600),
            child: Column(
              children: [
                Padding(
                  padding: const EdgeInsets.all(12),
                  child: ElevatedButton(
                    onPressed: _loading ? null : _loadCoins,
                    style: ElevatedButton.styleFrom(
                      backgroundColor: const Color(0xFF0055DD),
                      foregroundColor: Colors.white,
                      shape: RoundedRectangleBorder(
                        borderRadius: BorderRadius.circular(5),
                      ),
                    ),
                    child: Text(_loading ? 'Memuat...' : 'Refresh'),
                  ),
                ),
                if (_loading) const LinearProgressIndicator(),
                if (_error != null)
                  Padding(
                    padding: const EdgeInsets.all(12),
                    child: Text(_error!, style: const TextStyle(color: Colors.red)),
                  ),
                if (_error != null && _coins.isNotEmpty)
                  const Padding(
                    padding: EdgeInsets.only(bottom: 8),
                    child: Text('Menampilkan data terakhir yang berhasil dimuat.'),
                  ),
                Expanded(child: _buildList()),
              ],
            ),
          ),
        ),
      ),
    );
  }

  Widget _buildList() {
    if (_coins.isEmpty) {
      return Center(
        child: Text(_loading ? 'Mengambil data CoinLore...' :
          (_error == null ? 'Belum ada data cryptocurrency.' : 'Tekan Refresh untuk mencoba lagi.')),
      );
    }
    return RefreshIndicator(
      onRefresh: _loadCoins,
      child: ListView.builder(
        physics: const AlwaysScrollableScrollPhysics(),
        padding: const EdgeInsets.symmetric(horizontal: 8),
        itemCount: _coins.length,
        itemBuilder: (context, index) => _CoinRow(coin: _coins[index]),
      ),
    );
  }
}

class _CoinRow extends StatelessWidget {
  final Cryptocurrency coin;
  const _CoinRow({required this.coin});

  @override
  Widget build(BuildContext context) {
    return Container(
      padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 12),
      decoration: const BoxDecoration(
        color: Color(0xFFFFF2CC),
        border: Border(bottom: BorderSide(color: Color(0xFFD6B656))),
      ),
      child: Row(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Expanded(flex: 2, child: _field('Rank', coin.rank.toString())),
          const SizedBox(width: 8),
          Expanded(flex: 4, child: _field(coin.name, coin.symbol)),
          const SizedBox(width: 8),
          Expanded(flex: 5, child: _field('USD', coin.priceUsd)),
        ],
      ),
    );
  }

  Widget _field(String label, String value) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(label, style: const TextStyle(fontSize: 12)),
        const SizedBox(height: 3),
        Text(value, style: const TextStyle(fontSize: 20, fontWeight: FontWeight.w500)),
      ],
    );
  }
}

import 'dart:convert';
import 'package:http/http.dart' as http;
import '../models/cryptocurrency.dart';

class CryptoService {
  static final endpoint = Uri.parse('https://api.coinlore.net/api/tickers/');
  final http.Client _client;

  CryptoService({http.Client? client}) : _client = client ?? http.Client();

  Future<List<Cryptocurrency>> fetchCoins() async {
    final response = await _client.get(endpoint).timeout(
      const Duration(seconds: 20),
    );
    if (response.statusCode != 200) {
      throw Exception('Server mengembalikan HTTP ${response.statusCode}.');
    }
    final decoded = jsonDecode(response.body);
    if (decoded is! Map<String, dynamic> || decoded['data'] is! List) {
      throw const FormatException('Format respons API tidak sesuai.');
    }
    final coins = (decoded['data'] as List).map((item) {
      if (item is! Map<String, dynamic>) {
        throw const FormatException('Format item API tidak sesuai.');
      }
      return Cryptocurrency.fromJson(item);
    }).toList();
    coins.sort((a, b) => a.rank.compareTo(b.rank));
    return coins;
  }

  void dispose() => _client.close();
}

import 'package:flutter_test/flutter_test.dart';
import 'package:http/http.dart' as http;
import 'package:http/testing.dart';
import 'package:tugas03_crypto/models/cryptocurrency.dart';
import 'package:tugas03_crypto/services/crypto_service.dart';

void main() {
  test('Mengambil data dari endpoint dan mengurutkan berdasarkan rank', () async {
    final client = MockClient((request) async {
      expect(request.url.toString(), 'https://api.coinlore.net/api/tickers/');
      return http.Response('{"data":['
        '{"rank":2,"name":"Ethereum","symbol":"ETH","price_usd":"1720.54"},'
        '{"rank":1,"name":"Bitcoin","symbol":"BTC","price_usd":"27604.54"}'
        ']}', 200);
    });
    final service = CryptoService(client: client);
    addTearDown(service.dispose);
    final coins = await service.fetchCoins();
    expect(coins.first.rank, 1);
    expect(coins.first.name, 'Bitcoin');
    expect(coins.first.symbol, 'BTC');
    expect(coins.first.priceUsd, '27604.54');
    expect(coins.length, 2);
  });

  test('Harga koin kecil tetap memiliki presisi asli', () {
    final coin = Cryptocurrency.fromJson({
      'rank': '10', 'name': 'Contoh', 'symbol': 'TEST',
      'price_usd': '0.00000012',
    });
    expect(coin.priceUsd, '0.00000012');
  });

  test('Menolak field wajib yang hilang', () {
    expect(() => Cryptocurrency.fromJson({'rank': 1}), throwsFormatException);
  });

  test('Respons HTTP gagal menghasilkan exception', () async {
    final service = CryptoService(client: MockClient((_) async => http.Response('', 503)));
    addTearDown(service.dispose);
    await expectLater(service.fetchCoins(), throwsException);
  });

  test('Respons dengan struktur salah ditolak', () async {
    final service = CryptoService(client: MockClient((_) async => http.Response('{"data":{}}', 200)));
    addTearDown(service.dispose);
    await expectLater(service.fetchCoins(), throwsFormatException);
  });
}
